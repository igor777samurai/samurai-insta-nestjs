# START NEST PROJECT

nest new my-project
cd my-project

nest generate app api-gateway
nest generate app users
nest generate app posts

nest generate library contracts
npm install @nestjs/microservices @nestjs/cqrs

# DOCKER POSTGRES

docker compose logs postgres
docker exec -it nest-postgres psql -U app -d users_d
docker exec -it nest-postgres psql -U app -d users_db

# SQL POSTGRES

SELECT id, name, email, "passwordHash", "createdAt"
FROM "User"
ORDER BY id DESC;

# RUN

pnpm run start:dev users
pnpm run start:dev posts
pnpm run start:dev api-gateway
pnpm run start:dev auth
docker compose up -d
Stop-Service -Name postgresql-x64-18 // остановить локальный postgresql

# TEST

pnpm test
pnpm test:watch

# PRISMA (UI, MIGRATION)

pnpm exec prisma studio
pnpm exec prisma format
pnpm exec prisma validate
ls prisma/migrations
pnpm exec prisma migrate dev --name add_posts
pnpm exec prisma migrate dev --name add_auth_models
pnpm exec prisma generate

Get-Content prisma/schema.prisma
Get-Content prisma/migrations/20261024175549_init/migration.sql
Get-Content prisma/migrations/20261025040602_add_posts/migration.sql

# NEW MICROSERVICE

pnpm exec nest g app auth
